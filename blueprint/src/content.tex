\begin{definition}[Local field]
\label{def:local-field}

\begin{definition}[Function field]
\label{def:function-field}
\lean{AnalyticTheory.IsFunctionFieldOver}
\leanok
$F\,|\,K$ is a function field: finitely generated as a field extension, of transcendence degree $1$ over $K$, with $K$ algebraically closed in $F$.

\begin{definition}[Genus]
\label{def:genus}
\lean{AnalyticTheory.genus}
\uses{def:function-field, def:rr-space}
\leanok
The genus of $F\,|\,K$ is the least $g$ such that $\dim_K L(\mathfrak m)=\deg\mathfrak m-g+1$ for all divisors $\mathfrak m$ of sufficiently large degree (Riemann--Roch).

\label{def:conservative}
\lean{AnalyticTheory.IsConservative}
\uses{def:genus}
\leanok
$F\,|\,K$ is conservative if its genus does not change under every finite algebraic constant field extension $FL\,|\,L$ (a field generated by $F$ and $L$).
\end{definition}
